-- =========================================================
-- SUPERMARKET MANAGEMENT
-- PostgreSQL / Neon
-- Based strictly on the provided ERD
-- =========================================================


-- =========================================================
-- 0. DROP EVERYTHING IN PUBLIC SCHEMA
-- =========================================================

DROP SCHEMA public CASCADE;

CREATE SCHEMA public;


-- =========================================================
-- 1. CUSTOMER
-- =========================================================

CREATE TABLE CUSTOMER (
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    loyalty_points INTEGER,

    customer_id VARCHAR(36) PRIMARY KEY,

    full_name VARCHAR(100) NOT NULL,

    phone_number VARCHAR(15) UNIQUE,

    email VARCHAR(150)
);


-- =========================================================
-- 2. REVIEW
-- =========================================================

CREATE TABLE REVIEW (
    review_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    review_id VARCHAR(36) PRIMARY KEY,

    customer_id VARCHAR(36),

    rating INTEGER NOT NULL
        CHECK (rating >= 1 AND rating <= 5),

    content TEXT,

    CONSTRAINT FK_REVIEW_CUSTOMER
        FOREIGN KEY (customer_id)
        REFERENCES CUSTOMER(customer_id)
);


-- =========================================================
-- 3. EMPLOYEE
-- =========================================================

CREATE TABLE EMPLOYEE (
    position VARCHAR(30)
        DEFAULT 'CASHIER'
        CHECK (
            position IN (
                'STORE_OWNER',
                'CASHIER',
                'INVENTORY_MANAGER'
            )
        ),

    employee_id VARCHAR(36) PRIMARY KEY,

    full_name VARCHAR(100) NOT NULL,

    username VARCHAR(50) UNIQUE NOT NULL,

    password VARCHAR(255) NOT NULL,

    phone_number VARCHAR(15) UNIQUE NOT NULL,

    status BOOLEAN DEFAULT TRUE
);


-- =========================================================
-- 4. PRODUCT CATEGORY
-- =========================================================

CREATE TABLE PRODUCT_CATEGORY (
    category_id VARCHAR(36) PRIMARY KEY,

    category_name VARCHAR(100) NOT NULL
);


-- =========================================================
-- 5. PRODUCT
-- =========================================================

CREATE TABLE PRODUCT (
    product_status VARCHAR(30)
        DEFAULT 'ON_SALE'
        CHECK (
            product_status IN (
                'ON_SALE',
                'DISCONTINUED'
            )
        ),

    stock_quantity INTEGER NOT NULL,

    expiration_date TIMESTAMP NOT NULL,

    product_id VARCHAR(36) PRIMARY KEY,

    category_id VARCHAR(36),

    product_name VARCHAR(100) NOT NULL,

    unit_price numeric(12, 2) NOT NULL,

    CONSTRAINT FK_PRODUCT_CATEGORY
        FOREIGN KEY (category_id)
        REFERENCES PRODUCT_CATEGORY(category_id)
);


-- =========================================================
-- 6. INVOICE
-- =========================================================

CREATE TABLE INVOICE (
    created_at TIMESTAMP NOT NULL,

    status VARCHAR(30) NOT NULL
        CHECK (
            status IN (
                'PENDING',
                'PAID',
                'CANCELLED'
            )
        ),

    invoice_id VARCHAR(36) PRIMARY KEY,

    customer_id VARCHAR(36),

    employee_id VARCHAR(36),

    original_total DECIMAL(12,2) NOT NULL,

    discount_amount DECIMAL(12,2),

    CONSTRAINT FK_INVOICE_CUSTOMER
        FOREIGN KEY (customer_id)
        REFERENCES CUSTOMER(customer_id),

    CONSTRAINT FK_INVOICE_EMPLOYEE
        FOREIGN KEY (employee_id)
        REFERENCES EMPLOYEE(employee_id)
);


-- =========================================================
-- 7. INVOICE DETAIL
-- =========================================================

CREATE TABLE INVOICE_DETAIL (
    quantity INTEGER NOT NULL,

    invoice_detail_id VARCHAR(36) PRIMARY KEY,

    invoice_id VARCHAR(36),

    product_id VARCHAR(36),

    unit_price DECIMAL(12,2) NOT NULL,

    subtotal DECIMAL(12,2) NOT NULL,

    CONSTRAINT FK_INVOICE_DETAIL_INVOICE
        FOREIGN KEY (invoice_id)
        REFERENCES INVOICE(invoice_id),

    CONSTRAINT FK_INVOICE_DETAIL_PRODUCT
        FOREIGN KEY (product_id)
        REFERENCES PRODUCT(product_id)
);


-- =========================================================
-- 8. PAYMENT
-- =========================================================

CREATE TABLE PAYMENT (
    payment_time TIMESTAMP,

    status VARCHAR(30)
        DEFAULT 'UNPAID'
        CHECK (
            status IN (
                'UNPAID',
                'PAID',
                'FAILED'
            )
        ),

    payment_method VARCHAR(30)
        DEFAULT 'CASH'
        CHECK (
            payment_method IN (
                'CASH',
                'BANK_TRANSFER',
                'QR'
            )
        ),

    transaction_id VARCHAR(36) PRIMARY KEY,

    invoice_id VARCHAR(36),

    amount DECIMAL(12,2)
        CHECK (amount >= 0),

    CONSTRAINT FK_PAYMENT_INVOICE
        FOREIGN KEY (invoice_id)
        REFERENCES INVOICE(invoice_id)
);


-- =========================================================
-- 9. VOUCHER
-- =========================================================

CREATE TABLE VOUCHER (
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    voucher_id VARCHAR(36) PRIMARY KEY,

    employee_id VARCHAR(36),

    total_amount DECIMAL(12,2) NOT NULL,

    CONSTRAINT FK_VOUCHER_EMPLOYEE
        FOREIGN KEY (employee_id)
        REFERENCES EMPLOYEE(employee_id)
);


-- =========================================================
-- 10. PURCHASE RECEIPT
-- =========================================================

CREATE TABLE PURCHASE_RECEIPT (
    voucher_id VARCHAR(36) PRIMARY KEY,

    supplier_name VARCHAR(20) NOT NULL,

    CONSTRAINT FK_PURCHASE_RECEIPT_VOUCHER
        FOREIGN KEY (voucher_id)
        REFERENCES VOUCHER(voucher_id)
);


-- =========================================================
-- 11. PURCHASE RECEIPT DETAIL
-- =========================================================

CREATE TABLE PURCHASE_RECEIPT_DETAIL (
    purchased_quantity INTEGER NOT NULL
        CHECK (purchased_quantity > 0),

    expiration_date TIMESTAMP NOT NULL,

    purchase_detail_id VARCHAR(36) PRIMARY KEY,

    voucher_id VARCHAR(36),

    product_id VARCHAR(36),

    purchase_price DECIMAL(12,2)
        CHECK (purchase_price > 0),

    CONSTRAINT FK_PURCHASE_DETAIL_RECEIPT
        FOREIGN KEY (voucher_id)
        REFERENCES PURCHASE_RECEIPT(voucher_id),

    CONSTRAINT FK_PURCHASE_DETAIL_PRODUCT
        FOREIGN KEY (product_id)
        REFERENCES PRODUCT(product_id)
);


-- =========================================================
-- 12. DISPOSAL RECEIPT
-- =========================================================

CREATE TABLE DISPOSAL_RECEIPT (
    voucher_id VARCHAR(36) PRIMARY KEY,

    CONSTRAINT FK_DISPOSAL_RECEIPT_VOUCHER
        FOREIGN KEY (voucher_id)
        REFERENCES VOUCHER(voucher_id)
);


-- =========================================================
-- 13. DISPOSAL RECEIPT DETAIL
-- =========================================================

CREATE TABLE DISPOSAL_RECEIPT_DETAIL (
    disposed_quantity INTEGER NOT NULL,

    disposal_reason TEXT,

    disposal_detail_id VARCHAR(36) PRIMARY KEY,

    voucher_id VARCHAR(36),

    product_id VARCHAR(36),

    CONSTRAINT FK_DISPOSAL_DETAIL_RECEIPT
        FOREIGN KEY (voucher_id)
        REFERENCES DISPOSAL_RECEIPT(voucher_id),

    CONSTRAINT FK_DISPOSAL_DETAIL_PRODUCT
        FOREIGN KEY (product_id)
        REFERENCES PRODUCT(product_id)
);


-- =========================================================
-- 14. RETURN RECEIPT
-- =========================================================

CREATE TABLE RETURN_RECEIPT (
    requested_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    return_reason TEXT,

    return_receipt_id VARCHAR(36) PRIMARY KEY,

    employee_id VARCHAR(36) NOT NULL,

    original_invoice_id VARCHAR(36) NOT NULL,

    refund_amount DECIMAL(12,2),

    CONSTRAINT FK_RETURN_RECEIPT_EMPLOYEE
        FOREIGN KEY (employee_id)
        REFERENCES EMPLOYEE(employee_id),

    CONSTRAINT FK_RETURN_RECEIPT_INVOICE
        FOREIGN KEY (original_invoice_id)
        REFERENCES INVOICE(invoice_id)
);


-- =========================================================
-- 15. INVENTORY REPORT
-- =========================================================

CREATE TABLE INVENTORY_REPORT (
    report_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    report_id VARCHAR(36) PRIMARY KEY,

    employee_id VARCHAR(36),

    CONSTRAINT FK_INVENTORY_REPORT_EMPLOYEE
        FOREIGN KEY (employee_id)
        REFERENCES EMPLOYEE(employee_id)
);


-- =========================================================
-- 16. INVENTORY REPORT DETAIL
-- =========================================================

CREATE TABLE INVENTORY_REPORT_DETAIL (
    report_id VARCHAR(36) NOT NULL,

    product_id VARCHAR(36),

    system_quantity INTEGER
        CHECK (system_quantity >= 0),

    actual_quantity INTEGER
        CHECK (actual_quantity >= 0),

    CONSTRAINT FK_INVENTORY_DETAIL_REPORT
        FOREIGN KEY (report_id)
        REFERENCES INVENTORY_REPORT(report_id),

    CONSTRAINT FK_INVENTORY_DETAIL_PRODUCT
        FOREIGN KEY (product_id)
        REFERENCES PRODUCT(product_id)
);
